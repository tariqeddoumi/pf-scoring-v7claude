Option Explicit

Private Sub Workbook_Open()
    On Error Resume Next
    modOutil.ProtegerTout
    Application.Calculate
    modOutil.MajSynthese False
    Me.Worksheets(modOutil.ONGLET_ACCUEIL).Activate
End Sub

' Trace des changements de routage : ils modifient les seuils et les poids.
Private Sub Workbook_SheetChange(ByVal Sh As Object, ByVal Target As Range)
    On Error GoTo Fin
    If Not modOutil.EstDossier(Sh) Then Exit Sub
    If Not Intersect(Target, Sh.Range("B18:B20")) Is Nothing Then
        modOutil.Journaliser "Routage modifié", Sh.Name, _
            modOutil.Txt(Sh.Range("B18").Value) & " / " & modOutil.Txt(Sh.Range("B19").Value) & " / " & _
            modOutil.Txt(Sh.Range("B20").Value) & " -> note approuvée " & Sh.Range("B6").Text
        If Not Intersect(Target, Sh.Range("B19")) Is Nothing Then modOutil.SecteurModifie Sh
    End If
Fin:
End Sub
